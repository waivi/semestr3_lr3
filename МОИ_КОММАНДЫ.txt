# Создание и заполнение массива
./dbms --query "MARRAY test_arr MADDEND 'Первый элемент'"
./dbms --query "MARRAY test_arr MADDEND 'Второй элемент'"
./dbms --query "MARRAY test_arr MADDEND 'Третий элемент'"
./dbms --query "MARRAY test_arr MPRINT"

# Сохранение в текстовый файл
./dbms --query "SAVE ARRAY test_arr TEXT test_array.txt"

# Сохранение в бинарный файл
./dbms --query "SAVE ARRAY test_arr BINARY test_array.bin"

# Загрузка в новый массив
./dbms --query "LOAD ARRAY loaded_arr TEXT test_array.txt"
./dbms --query "MARRAY loaded_arr MPRINT"

./dbms --query "SAVE SLIST test_list BINARY test_slist.bin"
